    <script>
        const temas = [
            { t: "La Esperanza", c: "Hebreos 11:1", m: "La fe es la certeza de lo que se espera. Mantén tu esperanza viva hoy." },
            { t: "Paz Infinita", c: "Filipenses 4:7", m: "La paz de Dios sobrepasa todo entendimiento. Deja tus cargas en sus manos." },
            { t: "Amor Incondicional", c: "1 Juan 4:8", m: "Dios es amor. En cada detalle de tu día, busca su cariño protector." },
            { t: "Fortaleza Diaria", c: "Isaías 40:31", m: "Los que esperan en el Señor tendrán nuevas fuerzas." },
            { t: "Luz en el Camino", c: "Salmos 119:105", m: "Tu palabra es lámpara a mis pies y lumbrera a mi camino." }
        ];

        function obtenerDevocional(fecha) {
            const inicio = new Date(fecha.getFullYear(), 0, 0);
            const diff = fecha - inicio;
            const dia = Math.floor(diff / (1000 * 60 * 60 * 24));
            const tema = temas[dia % temas.length];
            return { titulo: tema.t, cita: tema.c, cuerpo: tema.m };
        }

        let fechaActual = new Date();
        let fontSize = 20;
        let grabacionActiva = false;
        const synth = window.speechSynthesis;
        let recognition = null;

        // Inicializar Reconocimiento con soporte para móviles
        const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
        if (SpeechRecognition) {
            recognition = new SpeechRecognition();
            recognition.lang = 'es-ES';
            recognition.continuous = false;
            recognition.interimResults = false;

            recognition.onstart = () => {
                grabacionActiva = true;
                actualizarBotonGrabacion(true);
                anunciar("Grabando. Hable ahora.");
            };

            recognition.onresult = (event) => {
                const texto = event.results[0][0].transcript;
                guardarNota(texto, 'voz');
            };

            recognition.onend = () => {
                grabacionActiva = false;
                actualizarBotonGrabacion(false);
            };

            recognition.onerror = (event) => {
                console.error(event.error);
                if(event.error === 'not-allowed') anunciar("Permiso de micrófono denegado.");
                else anunciar("Error al grabar.");
                grabacionActiva = false;
                actualizarBotonGrabacion(false);
            };
        }

        window.onload = () => {
            actualizarInterfaz();
            anunciar("Devocional optimizado para móvil listo.");
        };

        function gestionarDictado() {
            if (!recognition) {
                anunciar("Dictado no soportado en este móvil.");
                return;
            }
            try {
                if (grabacionActiva) recognition.stop();
                else recognition.start();
            } catch (e) {
                recognition.stop();
            }
        }

        function actualizarBotonGrabacion(estaGrabando) {
            const btn = document.getElementById('btn-dictar');
            const texto = document.getElementById('text-dictar');
            const status = document.getElementById('status-dictado');

            if (estaGrabando) {
                btn.classList.add('recording-active');
                texto.innerText = "Detener Grabación";
                status.innerText = "ESCUCHANDO...";
            } else {
                btn.classList.remove('recording-active');
                texto.innerText = "Toca para Grabar";
                status.innerText = "";
            }
        }

        function navegarFecha(dias) {
            fechaActual.setDate(fechaActual.getDate() + dias);
            actualizarInterfaz();
            anunciar(document.getElementById('fecha-display').innerText);
        }

        function irAHoy() {
            fechaActual = new Date();
            actualizarInterfaz();
            anunciar("Regresaste a hoy.");
        }

        function actualizarInterfaz() {
            const opciones = { weekday: 'short', day: 'numeric', month: 'short' };
            document.getElementById('fecha-display').innerText = fechaActual.toLocaleDateString('es-ES', opciones);
            const contenido = obtenerDevocional(fechaActual);
            document.getElementById('dev-titulo').innerText = contenido.titulo;
            document.getElementById('dev-cita').innerText = contenido.cita;
            document.getElementById('dev-cuerpo').innerText = contenido.cuerpo;
            cargarNotas();
        }

        function leerDevocional() {
            const t = document.getElementById('dev-titulo').innerText;
            const c = document.getElementById('dev-cita').innerText;
            const m = document.getElementById('dev-cuerpo').innerText;
            anunciar(`${t}. ${c}. ${m}`);
        }

        function anunciar(texto) {
            synth.cancel();
            const utterance = new SpeechSynthesisUtterance(texto);
            utterance.lang = 'es-ES';
            utterance.rate = 1.0;
            synth.speak(utterance);
        }

        function detenerVoz() {
            synth.cancel();
            if (grabacionActiva) recognition.stop();
        }

        function cambiarFuente(delta) {
            fontSize += delta;
            fontSize = Math.max(16, Math.min(50, fontSize));
            document.body.style.fontSize = fontSize + 'px';
            anunciar("Texto cambiado.");
        }

        function guardarEscritura() {
            const texto = document.getElementById('input-escritura').value.trim();
            if (texto) {
                guardarNota(texto, 'escrito');
                document.getElementById('input-escritura').value = '';
            }
        }

        function guardarNota(texto, tipo) {
            const iso = fechaActual.toISOString().split('T')[0];
            const db = JSON.parse(localStorage.getItem('dev_m_v3') || '{}');
            if (!db[iso]) db[iso] = [];
            db[iso].push({ id: Date.now(), texto, tipo, hora: new Date().toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'}) });
            localStorage.setItem('dev_m_v3', JSON.stringify(db));
            cargarNotas();
            anunciar("Nota guardada.");
        }

        function cargarNotas() {
            const iso = fechaActual.toISOString().split('T')[0];
            const db = JSON.parse(localStorage.getItem('dev_m_v3') || '{}');
            const notas = db[iso] || [];
            const lista = document.getElementById('lista-notas');
            lista.innerHTML = notas.length ? '' : '<p class="text-center opacity-50">Sin reflexiones hoy.</p>';
            
            notas.slice().reverse().forEach(n => {
                const div = document.createElement('div');
                div.className = 'note-card';
                div.innerHTML = `
                    <p class="text-yellow-500 font-bold mb-1">${n.tipo === 'voz' ? '🎤' : '✏️'} ${n.hora}</p>
                    <p class="text-xl mb-3">${n.texto}</p>
                    <div class="flex gap-4">
                        <button onclick="anunciar('${n.texto}')" class="text-blue-400 font-bold underline">Oír</button>
                        <button onclick="borrarNota(${n.id})" class="text-red-500 font-bold underline">Borrar</button>
                    </div>
                `;
                lista.appendChild(div);
            });
        }

        function borrarNota(id) {
            const iso = fechaActual.toISOString().split('T')[0];
            const db = JSON.parse(localStorage.getItem('dev_m_v3') || '{}');
            db[iso] = db[iso].filter(n => n.id !== id);
            localStorage.setItem('dev_m_v3', JSON.stringify(db));
            cargarNotas();
            anunciar("Borrado.");
        }
